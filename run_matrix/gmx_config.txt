# Configuration File for Molecular Dynamics Setup

# Box Type:
# Specify the type of box for simulation
# Options: Triclinic, Cubic, Dodecahedron, or Octahedron
box_type: Dodecahedron

# Distance:
# Specify the distance for the simulation
distance: 1.0

# Solvent Model:
# Specify the solvent for the simulation
solvent: spc216.gro

# Possible values for charge:
#   bcc   : AM1-BCC charges (the default)
#   gas   : Mulliken charges in gas phase
#   resp  : RESP charges
#   none  : No charges
charge: gas

# Possible values for atom_type:
#   gaff  : the default
#   gaff2 : for GAFF2 (beta-version)
#   amber : for PARM94/99/99SB
#   bcc   : for bcc
#   sybyl : for sybyl
atom_type: gaff2

# Protein force field for pdb2gmx (non-interactive)
forcefield: amber03

# Water model for pdb2gmx
water: tip3p

# Hydrogen Mass Repartitioning (pdb2gmx -heavyh): moves mass from heavy atoms
# into their bonded hydrogens, stabilizing a 4 fs timestep (requires
# constraints=all-bonds in *.mdp) instead of the usual 2 fs (h-bonds)
# ceiling -- roughly doubles production throughput. Only applied to the
# receptor via pdb2gmx; ligand topologies (antechamber/tleap) are unaffected,
# which is fine since LINCS with all-bonds constrains every ligand bond
# regardless of atom mass, and water uses SETTLE rather than LINCS.
heavy_hydrogens: no

# Name of the merged Protein+ligand index group used in mdp tc-grps
merged_group: Protein_LIG

# Extra options passed to gmx mdrun (resource selection).
# Left empty by default so GROMACS auto-detects the available hardware; this
# is what you want on an unfamiliar machine. Set it explicitly to pin
# resources, for example on a GPU workstation:
#   mdrun_options: -ntmpi 1 -ntomp 16 -nb gpu
# On a GPU machine -ntmpi must be given whenever -ntomp/OMP_NUM_THREADS is set.
mdrun_options:

# Net charge passed to antechamber (-nc). "auto" reads the formal charge
# column of the input PDB, or sums the partial charges of a MOL2 file.

# Optional structure curation with pdbfixer (receptors here are pre-curated)
# Path to a Python interpreter that has pdbfixer/OpenMM installed. Leave
# empty to use the interpreter running LAGMX itself. Only consulted when
# fix_structure is "yes"; see environment.yml for how to create such an env.
fix_structure: no
fixer_python:

# Fallback SEQRES source for structure curation, used only when a receptor
# file has no SEQRES records of its own and no matching ref_*.pdb sits next
# to it in the complex directory. Must be >=90% identical to the observed
# residues or it is rejected (see attach_seqres()/sequence_identity()).
seqres_reference:

# Target production MD length in nanoseconds. Replaces manually editing
# `nsteps` in md.mdp.
#   - Complex belum punya hasil produksi -> dijalankan dari 0 ns sampai
#     production_ns (perilaku normal).
#   - Complex SUDAH punya md.cpt dari run sebelumnya (mis. selesai 1 ns) dan
#     production_ns lebih besar dari itu -> OTOMATIS dilanjutkan dari
#     checkpoint sampai production_ns (system-prep & equilibrasi di-skip,
#     hasil lama tidak dihapus/diulang).
#   - Complex sudah mencapai production_ns -> di-skip.
#   - Kosongkan value ini untuk kembali ke perilaku lama (pakai nsteps
#     bawaan md.mdp, selalu mulai dari 0 ns).
production_ns: 5

# Stop after NPT equilibration and skip the production mdrun entirely.
# Useful to prepare systems (prep+EM+NVT+NPT, all cheap and short) ahead of
# time and defer the expensive, timed production run until the machine is
# actually idle -- set to "no" and re-run to resume automatically from the
# existing NPT checkpoint straight into production.
skip_production: no

# ---------------------------------------------------------------------------
# Analisis pasca-produksi (analyze_md.py)
#
# Dibaca hanya oleh analyze_md.py; LAGMX.py mengabaikannya. Seluruh key di
# bawah ini opsional — config lama tanpa key ini tetap jalan dengan nilai
# bawaan yang sama seperti tertulis di sini.
#
#   cd run_matrix && python3 ../analyze_md.py
# ---------------------------------------------------------------------------

# Analisis yang dijalankan, dipisah koma. "all" menjalankan semuanya
# termasuk mmpbsa. Pilihan:
#   rmsd      stabilitas kompleks (backbone protein dan ligan terpisah)
#   rmsf      fleksibilitas per residu
#   rg        kompaktibilitas protein
#   sasa      luas permukaan protein, ligan, kompleks, dan area terkubur
#   hbond     ikatan hidrogen protein-ligan sepanjang lintasan
#   contacts  okupansi kontak per residu -> residu kunci
#   pca       gerakan global, proyeksi PC1/PC2
#   fel       free energy landscape di atas proyeksi PCA
#   mmpbsa    energi bebas ikatan lewat gmx_MMPBSA (butuh mmpbsa_python)
analysis: rmsd,rmsf,rg,sasa,hbond,contacts,pca,fel

# Berapa nanodetik pertama dibuang sebelum analisis. Awal produksi masih
# memuat relaksasi sisa dari ekuilibrasi; memasukkannya menggeser rerata RMSD
# dan mengaburkan plateau. Nol berarti seluruh lintasan dipakai.
analysis_skip_ns: 0

# Ambang kontak protein-ligan dalam nm. 0.4 nm adalah konvensi yang lazim
# dipakai untuk kontak non-ikatan.
analysis_contact_cutoff: 0.4

# gmx mana yang dipakai untuk analisis. Boleh berbeda dari gmx yang dipakai
# menjalankan simulasi — analisis tidak butuh GPU, jadi build apa pun cukup.
analysis_gmx: gmx

# Penjarangan frame per analisis: pakai satu frame dari setiap N. Nilai 1
# berarti semua frame dipakai.
#
# Setiap alat analisis gmx berjalan satu thread dan membaca lintasan dari awal
# sampai akhir — tidak ada opsi -nt untuk dilempari core. Jadi biayanya sebanding
# dengan jumlah frame, dan tidak dengan apa pun yang lain. Produksi 100 ns yang
# ditulis tiap 2 ps menghasilkan 45.000 frame; pada satu kompleks ThiM nyata itu
# berarti 5,4 jam untuk empat analisis, 3,4 jam di antaranya hanya untuk SASA.
#
# Penjarangan bukan jalan pintas, melainkan pernyataan tentang skala waktu tiap
# besaran. Frame berjarak 2 ps sangat berkorelasi: hampir sampel yang sama
# dihitung berulang, jadi menambah waktu tanpa menambah informasi. Nilai bawaan
# di bawah ini mengikuti dua hal — berapa mahal analisisnya, dan skala waktu apa
# yang sebenarnya bisa ia lihat.
#
# rmsd, rmsf, rg  murah (10–19 menit); menjarangkan menghemat sedikit sekali,
#                 sementara ketiganya justru dibaca frame demi frame.
analysis_stride_rmsd: 1
analysis_stride_rmsf: 1
analysis_stride_rg: 1
# sasa            paling mahal. Luas permukaan terpapar pelarut berubah lambat;
#                 4.500 sampel sepanjang 90 ns sudah menampilkan segalanya.
analysis_stride_sasa: 10
# hbond           hanya statistik okupansi — skrip meminta -num saja, tidak
#                 pernah lifetime atau autokorelasi. Tetap dibuat lebih halus
#                 dari sasa karena ikatan hidrogen putus-sambung pada skala
#                 ps–ns dan bentuk kurvanya perlu terjaga. Menaikkan nilai ini
#                 aman hanya selama belum ada analisis lifetime.
analysis_stride_hbond: 5
# contacts        okupansi per residu adalah sebuah fraksi; 4.500 frame menekan
#                 galat bakunya jauh di bawah satu persen.
analysis_stride_contacts: 10
# pca             kovarians butuh sampel independen, dan frame 2 ps bukan itu.
#                 Penjarangan memperbaiki kondisi matriksnya, bukan merusaknya,
#                 sekaligus mengecilkan eigenvec.trr dengan faktor yang sama.
analysis_stride_pca: 10
# fel             tidak ada di sini: gmx sham membin proyeksi PCA dan tidak
#                 pernah membuka lintasan, jadi ia mewarisi penjarangan pca.
# mmpbsa          punya knob sendiri, mmpbsa_interval, dengan alasan yang sama.

# gmx_MMPBSA hidup di environment terpisah karena menarik AmberTools versi
# sendiri. Isi dengan salah satu dari: prefix environment conda, interpreter
# python di dalamnya, atau path langsung ke executable gmx_MMPBSA. Dikosongkan
# berarti analisis mmpbsa dilewati, bahkan bila diminta di key `analysis`.
mmpbsa_python:

# gb, pb, atau both. MM/GBSA jauh lebih murah dan biasanya cukup untuk
# memeringkat ligan; MM/PBSA lebih mahal dan lebih tepat secara absolut.
mmpbsa_method: gb

# Dua cara menentukan frame mana yang disampling. `mmpbsa_interval` menang
# bila diisi.
#
# mmpbsa_interval: jarak antar frame, dalam satuan frame. Ini yang punya makna
#   fisik: frame yang berjarak lebih dekat daripada waktu autokorelasi sistem
#   menambah biaya tanpa menambah informasi, dan galat standar yang dihitung
#   atasnya tampak lebih kecil daripada yang sebenarnya karena rumusnya
#   mengasumsikan sampel independen.
#
#   Dengan nstxout-compressed 1000 dan dt 0.002 (satu frame tiap 2 ps),
#   pada produksi 100 ns dikurangi 10 ns awal = 45.000 frame tersisa:
#
#        interval    jarak      frame disampling
#              50    100 ps                 900
#             100    200 ps                 450
#             250    500 ps                 180   <- lazim dipakai
#             500    1,0 ns                  90
#
# mmpbsa_frames: dipakai hanya bila mmpbsa_interval kosong — jumlah frame yang
#   dituju, dan intervalnya dihitung dari panjang lintasan.
mmpbsa_interval: 250
mmpbsa_frames: 100

# Berapa proses MPI untuk gmx_MMPBSA. 1 = serial.
#
# gmx_MMPBSA membagi *frame* antar rank, jadi ini satu-satunya tuas yang membuat
# tahap ini selesai lebih cepat — dan tahap ini yang termahal. Diukur pada satu
# kompleks ThiM 75.852 atom dengan 181 frame: 98 menit serial, di mesin yang 40
# core-nya menganggur.
#
# Lebih banyak rank daripada frame hanya pemborosan; skrip membatasinya sendiri.
# `mpirun` diambil dari environment gmx_MMPBSA (mmpbsa_python), bukan dari PATH
# sistem: mencampur launcher sistem dengan mpi4py bawaan conda menghasilkan rank
# yang semuanya mengira dirinya rank 0, sehingga frame yang sama dihitung
# berulang tanpa satu pun pesan galat.
mmpbsa_np: 1

# Model Generalized Born (igb) dan konsentrasi garam dalam M.
mmpbsa_igb: 5
mmpbsa_salt: 0.150
